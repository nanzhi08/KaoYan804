2．有12个数围成一圈，求出相邻三个数之和的最小值。
int min(int *a,int n,int *minsum)
{
    int num,i,sum;   
    *minsum=*a+*(a+1)+*(a+2); 
    for(i=1; i<n; i++) 
    {
        sum=*(a+i)+*(a+(i+1)%n)+*(a+(i+2)%n);
        if(sum<*minsum)
        {
            num=i;
            *minsum=sum; 
        }
    }
    return num;
}

{
