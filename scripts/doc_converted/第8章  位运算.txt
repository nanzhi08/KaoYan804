二．练习内容
1.  编写一个函数getbits，从一个16位的单元中取出某几位（即该几位保留原值，其余位为0）。函数调用形式为：getbits(value,n1,n2)，其中value为该16位（二个字节）中的数据值，n1为欲取出的起始位，n2为欲取出的结束位。如：getbits(0101675,5,8)表示对八进制数0101675取出它的从左面开始的第5位到第8位。
#include "stdio.h"
int getbits(int value, int n1, int n2) 
 { 
 int mask=1; 
 int i;
 int t; 
 if(n2<n1) 
 { 
 t=n2; 
 n2=n1; 
 n1=t; 
 } 
 int len = n2-n1; 
 for(i=0;i<len;i++) 
 { 
 mask<<= 1;
 mask|= 1; 
 } 
 mask<<=16-n2;
 value&=mask; 
 value>>=16-n2;
 return value; 
 } 
main()
{
     int a;
    int n1,n2;
    printf("input an octal number:");
    scanf("%o",&a);
    printf("input n1,n2:");
    scanf("%d,%d",&n1,&n2);
    printf("result:%o\n",getbits(a,n1,n2));
getch();
}
2.  编写一个函数getbits(value)，从一个16位的二进制数中取出它的奇数位（即从左边起的第1、3、5、．．．、15位）。参数value为该16位（二个字节）中的数据值。
#include "stdio.h"
void main()
{
    unsigned short getbits(unsigned short value);
    unsigned short int a;
    printf("请输入一个八进制数:");
    scanf("%o",&a);
    printf("结果:%o\n",getbits(a));
    getch();
}
 
unsigned short getbits(unsigned short value)
{
    int i,j;
    unsigned short int z,a,q;
    z=0;
    for(i=1;i<=15;i+=2)
    {
        q=1;
        for(j=1;j<=(16-i-1)/2;j++) 
            q=q*2;
        a=value>>(16-i);
        a=a<<15; 
        a=a>15;
        z=z+a*q;
    }
    return(z);
}
